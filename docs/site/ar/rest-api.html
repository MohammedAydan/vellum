نظرة عامة على REST API#

واجهة HTTP مُركّبة تحت /v1 وتُرجع JSON في كل استجابة، مع إرفاق X-Request-Id. تتم مصادقة نقاط نهاية مستوى المستندات بالترويسة x-project-key؛ أما نقاط نهاية Console فتستخدم رمز جلسة Bearer. تتوفر وثيقة OpenAPI 3.1 قابلة للقراءة آليًا على GET /v1/openapi.json.

NOTE

فضّل SDK لكود التطبيقات. استخدم HTTP الخام عند الاستدعاء من لغة لا يوجد لها SDK رسمي أو عند تصحيح الأخطاء. ويوجد المرجع الكامل لنقطة نهاية بنقطة نهاية في مرجع HTTP API.

السيناريو#

أنت تدمج VELLUM من لغة لا يتوفر لها SDK رسمي. تكتشف نقاط النهاية المتاحة من /v1/openapi.json، ثم تُجري طلبًا موقّعًا بـ x-project-key لقراءة مستند وطلب POST إلى /query لجلب صفحة مُصفّاة — مع فحص X-Request-Id في كل استجابة للدعم الفني.

BASH
# اكتشف العقد الحي قبل كتابة كود التكامل.
curl -fsS "https://api.example.com/v1/openapi.json" | jq '.paths | keys[]'

طلب خام#

BASH
curl "https://api.example.com/v1/projects/<PROJECT_ID>/documents/users/u_123" \
  -H "x-project-key: <YOUR_API_KEY>"
JSON

بالنسبة إلى الاستعلام، أرسل POST بمواصفة استعلام إلى .../query؛ تكون الاستجابة { documents, nextCursor, meta }:

BASH
curl -X POST "https://api.example.com/v1/projects/<PROJECT_ID>/query" \
  -H "x-project-key: <YOUR_API_KEY>" \
  -H "content-type: application/json" \
  -d '{"collectionPath":"users","where":[{"field":"status","op":"==","value":"active"}],"limit":25}'

راجع الاستعلام والترقيم الصفحي.

WARNING

لا تضع مفتاح خادم أو رمز جلسة في URL ولا تُلتزم به في التحكم بالمصادر. استخدم عناصر نائبة ومرّر بيانات الاعتماد في الترويسات من مخزن أسرار.

الأخطاء الشائعة#

  • وضع المفتاح في URL. أرسل x-project-key (أو Authorization: Bearer) في ترويسة دائمًا.
  • الخلط بين واجهتي المصادقة. نقاط نهاية المستندات تأخذ مفتاح API؛ ونقاط نهاية Console تأخذ رمز جلسة Bearer.
  • نسيان X-Request-Id. التقطه في كل استجابة ليتمكن الدعم من ربط الحوادث بأمان.
  • تخطي مواصفة OpenAPI. GET /v1/openapi.json هو مخطط الطلب/الاستجابة الموثوق والمُحدَّث.

مثال موثّق#

المجموعة / البروتوكول
DART
import 'dart:io';
// Executed against the live PostgreSQL API by scripts/run-dart-examples.mjs.
//
// Raw REST API — call the public data plane with `vellum_flutter`'s request()
// helper, which attaches the x-project-key header for you.
// What you'll build: document POST/GET/PUT/PATCH/DELETE, a collection listing
// and a structured /query call.
// Real-world scenario: request() is the escape hatch for endpoints that have no
// convenience method (indexes, rulesets, batchWrite, transactions, etc.).
import 'package:vellum_flutter/vellum_flutter.dart';

Future<void> main() async {
  final endpoint = Platform.environment['VELLUM_API_BASE'] ??
      const String.fromEnvironment('VELLUM_API_BASE',
          defaultValue: 'http://127.0.0.1:8080');
  final projectId = Platform.environment['VELLUM_PROJECT_ID'] ??
      const String.fromEnvironment('VELLUM_PROJECT_ID');
  final apiKey = Platform.environment['VELLUM_API_KEY'] ??
      const String.fromEnvironment('VELLUM_API_KEY');

  final client =
      VellumClient(endpoint: endpoint, projectId: projectId, apiKey: apiKey);
  final prefix = 'ex_flutter_rest_api_${DateTime.now().microsecondsSinceEpoch}';
  final docPath = '$prefix/rest-doc';

  try {
    print('=== Raw REST API (Flutter) ===');

    await client.request('POST', '/documents/$docPath', body: {
      'data': {'title': 'raw', 'count': 1}
    });
    print('POST /documents/:path: 201');

    final fetched = await client.request('GET', '/documents/$docPath');
    print('GET: version=${fetched['version']}');

    final put = await client.request('PUT', '/documents/$docPath', body: {
      'data': {'title': 'merged'},
      'merge': true
    });
    print('PUT: version=${put['version']}');

    final patch = await client.request('PATCH', '/documents/$docPath', body: {
      'data': {'count': 2}
    });
    print('PATCH: version=${patch['version']}');

    final query = await client.request('POST', '/query', body: {
      'collectionPath': prefix,
      'where': [
        {'field': 'count', 'op': '==', 'value': 2},
      ],
      'orderBy': [],
      'limit': 10,
      'select': null,
      'startAt': null,
      'startAfter': null,
    });
    final ids =
        (query['documents'] as List).map((d) => d['documentId']).toList();
    print('POST /query: ids=$ids');

    final listing = await client.request('GET', '/documents/$prefix?limit=10');
    print('GET collection: count=${(listing['documents'] as List).length}');

    await client.request('DELETE', '/documents/$docPath');
    print('DELETE: 200');
  } finally {
    try {
      await client.doc(docPath).delete();
    } catch (_) {}
    client.dispose();
  }
}

/**
 * Raw REST API — call the public data plane with `fetch` directly.
 *
 * What you'll build: the same document lifecycle as the SDK examples, but using
 * plain HTTP and the x-project-key header (no SDK involved).
 * Real-world scenario: a non-JavaScript client or a thin serverless function
 * can talk to Vellum with nothing but fetch. This covers POST/GET/PUT/PATCH/
 * DELETE, a collection listing and the structured /query endpoint.
 */
import { harness, check } from "./_harness.js";

const h = await harness("rest-api");
const col = h.prefix;
const docPath = `${col}/rest-doc`;
const dataBase = `/v1/projects/${h.projectId}`;
const auth = { "x-project-key": h.apiKey };

try {
  h.log("=== Raw REST API ===");

  const created = await h.api.raw("POST", `${dataBase}/documents/${docPath}`, {
    data: { title: "raw", count: 1 },
  }, auth);
  check(created.status === 201, "POST created the document");
  h.log({ "POST /documents/:path": { status: created.status, id: (created.body as { documentId?: string }).documentId } });

  const fetched = await h.api.raw("GET", `${dataBase}/documents/${docPath}`, undefined, auth);
  const version = (fetched.body as { version: number }).version;
  check(fetched.status === 200 && version === 1, "GET returned version 1");
  h.log({ "GET /documents/:path": { status: fetched.status, version } });

  const put = await h.api.raw("PUT", `${dataBase}/documents/${docPath}`, {
    data: { title: "merged" },
    merge: true,
  }, auth);
  check(put.ok, "PUT with merge succeeded");
  h.log({ "PUT /documents/:path": { status: put.status, version: (put.body as { version: number }).version } });

  const patch = await h.api.raw("PATCH", `${dataBase}/documents/${docPath}`, {
    data: { count: 2 },
  }, auth);
  check(patch.ok, "PATCH succeeded");
  h.log({ "PATCH /documents/:path": { status: patch.status, version: (patch.body as { version: number }).version } });

  const query = await h.api.raw("POST", `${dataBase}/query`, {
    collectionPath: col,
    where: [{ field: "count", op: "==", value: 2 }],
    orderBy: [],
    limit: 10,
    select: null,
    startAt: null,
    startAfter: null,
  }, auth);
  const ids = ((query.body as { documents?: Array<{ documentId: string }> }).documents ?? []).map((d) => d.documentId);
  check(query.ok && ids.includes("rest-doc"), "query found the document");
  h.log({ "POST /query": { status: query.status, ids } });

  const listing = await h.api.raw("GET", `${dataBase}/documents/${col}?limit=10`, undefined, auth);
  check(listing.ok, "collection listing succeeded");
  h.log({ "GET /documents/:collection": { status: listing.status, count: (listing.body as { documents?: unknown[] }).documents?.length } });

  const removed = await h.api.raw("DELETE", `${dataBase}/documents/${docPath}`, undefined, auth);
  check(removed.ok, "DELETE succeeded");
  h.log({ "DELETE /documents/:path": { status: removed.status } });
} finally {
  await h.cleanup();
}
