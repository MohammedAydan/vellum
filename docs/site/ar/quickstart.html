البداية / البدء السريع

البدء السريع#

يُدخلك هذا الدليل في عالم VELLUM خلال دقائق قليلة: كتابة مستند وقراءته مجددًا. نستخدم SDK الرسمي بلغة TypeScript/JavaScript وهو @vellum/sdk-js.

السيناريو#

لقد أنشأت مشروعًا جديدًا في لوحة التحكم وتريد التأكد من أن الإعداد يعمل من طرف إلى طرف: ثبّت SDK، وأنشئ عميلًا بمفتاح إنتاج، واكتب مستند مستخدم واحدًا، واقرأه مجددًا، ثم احذفه. في نهاية هذا الدليل سيكون لديك عميل عامل يمكنك توسيعه ل بقية ميزات SDK. المشروع هو المستأجر الوحيد.

1. ثبّت SDK#

BASH
npm install @vellum/sdk-js

الحزمة هي ESM (type: module) وتستهدف Node 22+ أو متصفحًا حديثًا. وتتضمن تعريفات TypeScript الخاصة بها، لذا لا حاجة إلى حزمة @types منفصلة.

2. احصل على مفتاح مشروع#

يجب على طلب مستوى المستندات تقديم مفتاح API في الترويسة x-project-key. أنشئ مفتاحًا في لوحة التحكم ضمن بيئة مشروعك، أو عبر Console API. عامل مفاتيح الخادم (sk_) كأسرار؛ يمكن شحن المفاتيح العامة (pk_) إلى المتصفحات وتطبيقات الجوال لكنها لا تمنح أي وصول ضمني — تُفحص كل عملية مقابل قواعد أمان البيئة.

لا تلتزم بمفتاح حقيقي في التحكم بالمصادر. استخدم عنصرًا نائبًا مثل <YOUR_API_KEY> في الأمثلة، وحمّل القيمة الحقيقية من متغير بيئة وقت التشغيل.

3. أنشئ عميلًا#

TS
import { createClient } from "@vellum/sdk-js";

const db = createClient({
  endpoint: process.env.VELLUM_ENDPOINT ?? "http://localhost:8080",
  projectId: process.env.VELLUM_PROJECT_ID ?? "proj_01JCK9X2P8", // اختياري؛ الافتراضي production (الأسماء القديمة مقبولة)
  apiKey: process.env.VELLUM_API_KEY ?? "<YOUR_API_KEY>",
  // اختياري: قدّم JWT للمستخدم النهائي حتى تتمكن القواعد من تقييم auth.uid / auth.claims.
  getAuthToken: async () => currentUserJwt(),
});

يتطلب createClient كلًا من projectId وapiKey. تكون endpoint افتراضيًا http://localhost:8080. يمكن لتحديد realtimeUrl اختياريًا تجاوز عنوان WebSocket المشتق <endpoint>/v1/realtime، أما customFetch / webSocketFactory فهما نقاط اختبار (test seams). تُرسل جميع الطلبات إلى /v1/projects/:projectId/....

4. اكتب واقرأ واحذف#

TS
const users = db.collection("users");
const user = users.doc("user_123");

await user.create({ name: "Ada", role: "admin", active: true });
const snapshot = await user.get();
console.log(snapshot.id, snapshot.version, snapshot.data());

await user.update({ active: false });
await user.delete();
  • يفشل create إذا كان المستند موجودًا بالفعل (ALREADY_EXISTS).
  • يدمج set(data, { merge: true }) بعمق؛ وبدون merge يستبدل المستند.
  • يعدّل update الحقول المحددة ويمكن أن يحمل تحويلات حقول ذرية.
  • يكتب delete علامة حذف (tombstone) وحدثًا في صندوق البريد؛ إنه حذف ناعم.

بمجرد وجود مستند، يمكنك الاستعلام عن المجموعة مباشرة:

TS
// اقرأ جميع المستخدمين النشطين، عشرة في كل مرة.
const active = await users.where("active", "==", true).limit(10).get();
for (const doc of active.docs) console.log(doc.id, doc.data());

راجع CRUD لقواعد المسارات وتحويلات الحقول لـ fieldOp.*.

إعداد البيئة#

يقرأ خادم API التهيئة من متغيرات البيئة (ومن ملف .env اختياري). المفيدة للتطوير المحلي:

المتغيرالمعنىالافتراضي
DATABASE_MODEmemory للاختبارات المعزولة، postgres للتخزين الدائمmemory
CONTROL_DATABASE_URLاتصال PostgreSQL بمستوى التحكمpostgres://vellum:vellum_dev_password@localhost:5432/control_db
DATA_DATABASE_URLاتصال PostgreSQL بمستوى البياناتpostgres://vellum:vellum_dev_password@localhost:5433/data_db
DATA_DATABASE_URL_LISTENعنوان وضع الجلسة المحجوز لعميل LISTEN الخاص بالوقت الفعليغير محدد (وقت فعلي بالاستطلاع فقط)
API_PORT / API_HOSTعنوان الاستماع8080 / 0.0.0.0
CURSOR_HMAC_SECRETيوقّع مؤشرات الترقيم الصفحيالافتراضي للتطوير (يجب استبداله في الإنتاج)
WARNING

في NODE_ENV=production يرفض الخادم التشغيل إذا كان أي سر تطوير مخبوء لا يزال مستخدمًا. اضبط قيمًا قوية وفريدة لـ BETTER_AUTH_SECRET وAPI_KEY_PEPPER وCURSOR_HMAC_SECRET وJWT_SECRET قبل النشر. لا تسجّل أبدًا عناوين الاتصال أو قيم الأسرار.

import 'dart:io';
// Executed against the live PostgreSQL API by scripts/run-dart-examples.mjs.
//
// Quickstart — verify the connection and run one document lifecycle with
// `vellum_flutter`.
// What you'll build: a status-page service record that counts views.
// Real-world scenario: before building a screen, confirm the client can create,
// read, patch and delete a document. A short-lived server key is injected via
// `--dart-define=VELLUM_API_KEY=...`; no secret is hard-coded here.
import 'package:vellum_flutter/vellum_flutter.dart';

Future<void> main() async {
  final endpoint = Platform.environment['VELLUM_API_BASE'] ??
      const String.fromEnvironment('VELLUM_API_BASE',
          defaultValue: 'http://127.0.0.1:8080');
  final projectId = Platform.environment['VELLUM_PROJECT_ID'] ??
      const String.fromEnvironment('VELLUM_PROJECT_ID');
  final apiKey = Platform.environment['VELLUM_API_KEY'] ??
      const String.fromEnvironment('VELLUM_API_KEY');

  final client = VellumClient(
    endpoint: endpoint,
    projectId: projectId,
    apiKey: apiKey,
  );

  final prefix =
      'ex_flutter_quickstart_${DateTime.now().microsecondsSinceEpoch}';
  final ref = client.collection(prefix).doc('status-page');

  try {
    print('=== Quickstart (Flutter) ===');

    await ref
        .set({'service': 'checkout-api', 'status': 'operational', 'views': 0});
    final created = await ref.get();
    print('created: exists=${created.exists} version=${created.version}');

    await ref.update({'views': 1});
    final updated = await ref.get();
    final data = updated.data() ?? const <String, dynamic>{};
    print('updated: version=${updated.version} views=${data['views']}');

    await ref.delete();
    print('deleted: true');
  } finally {
    try {
      await ref.delete();
    } catch (_) {
      // Best effort cleanup; the doc may already be gone.
    }
    client.dispose();
  }
}

المصادقة — بما في ذلك المفاتيح العامة/الخاصة ورموز المستخدم النهائي — موضحة في القسم التالي.

المجموعة / البروتوكول
DART
import 'dart:io';
// Executed by scripts/run-dart-examples.mjs against a PostgreSQL API.
//
// Authentication & API keys — full key lifecycle with immediate revocation.
// What you'll build: sign-in, session check, server key creation, a probe with
// that key, revocation, and proof the revoked key is rejected immediately.
// Real-world scenario: rotating a leaked key must take effect on the next
// request. Auth/console endpoints are outside baseUrl, so httpTransport is used
// directly; credentials are injected via --dart-define, never hard-coded.
import 'package:vellum_flutter/vellum_flutter.dart';

Future<void> main() async {
  final endpoint = Platform.environment['VELLUM_API_BASE'] ??
      const String.fromEnvironment('VELLUM_API_BASE',
          defaultValue: 'http://127.0.0.1:8080');
  final projectId = Platform.environment['VELLUM_PROJECT_ID'] ??
      const String.fromEnvironment('VELLUM_PROJECT_ID');
  final apiKey = Platform.environment['VELLUM_API_KEY'] ??
      const String.fromEnvironment('VELLUM_API_KEY');
  final email = Platform.environment['VELLUM_EMAIL'] ??
      const String.fromEnvironment('VELLUM_EMAIL');
  final password = Platform.environment['VELLUM_PASSWORD'] ??
      const String.fromEnvironment('VELLUM_PASSWORD');

  final client =
      VellumClient(endpoint: endpoint, projectId: projectId, apiKey: apiKey);
  final prefix = 'ex_flutter_auth_${DateTime.now().microsecondsSinceEpoch}';
  final keysUrl = '$endpoint/v1/console/projects/$projectId/api-keys';
  final probeUrl = '$endpoint/v1/projects/$projectId/documents/$prefix/probe';
  final jsonHeaders = {'Content-Type': 'application/json'};

  try {
    print('=== Authentication & API Keys (Flutter) ===');

    final login = await client.httpTransport(
      'POST',
      '$endpoint/v1/auth/sign-in/email',
      jsonHeaders,
      {'email': email, 'password': password},
    );
    final loginMap = objectMap(login, 'sign-in');
    final token = loginMap['token'];
    if (token is! String || token.isEmpty)
      throw StateError('Missing sign-in token');
    print('signIn: success=${loginMap['success']} tokenLength=${token.length}');

    final session = await client.httpTransport(
      'GET',
      '$endpoint/v1/auth/session',
      {...jsonHeaders, 'Authorization': 'Bearer $token'},
      null,
    );
    print(
        'session: hasSession=${objectMap(session, 'session')['session'] != null}');

    final authHeaders = {...jsonHeaders, 'Authorization': 'Bearer $token'};
    final listedResponse =
        await client.httpTransport('GET', keysUrl, authHeaders, null);
    final listed = listedResponse is VellumHttpResponse
        ? listedResponse.body
        : listedResponse;
    if (listed is! List) throw StateError('Expected API key list');
    for (final key in listed) {
      objectMap(key, 'API key');
    }
    print('listKeys: responseReceived=${listed.isNotEmpty}');

    final created = await client.httpTransport('POST', keysUrl, authHeaders, {
      'name': 'examples:flutter-auth:ephemeral',
      'keyType': 'server',
    });
    final createdMap = objectMap(created, 'created key');
    final ephemeralId = objectMap(createdMap['apiKey'], 'API key')['id'];
    final rawKey = createdMap['rawKey'];
    if (ephemeralId is! String || rawKey is! String)
      throw StateError('Malformed created key');
    print('createKey: idReturned=${ephemeralId.isNotEmpty}');

    try {
      var missingKeyRejected = false;
      try {
        await client.httpTransport('GET', probeUrl, jsonHeaders, null);
      } on VellumException catch (e) {
        if (e.statusCode != 401) rethrow;
        missingKeyRejected = true;
        print('missingKey: code=${e.code} status=${e.statusCode}');
      }

      if (!missingKeyRejected) throw StateError('Missing key was accepted');
      try {
        await client.httpTransport(
            'GET', probeUrl, {...jsonHeaders, 'x-project-key': rawKey}, null);
      } on VellumException catch (error) {
        // A missing probe document still proves that authentication succeeded.
        if (error.statusCode != 404) rethrow;
      }
      print('freshKeyAuthenticates: true');
    } finally {
      await client.httpTransport(
          'DELETE', '$keysUrl/$ephemeralId', authHeaders, null);
    }
    print('revokeKey: true');
    var revokedKeyRejected = false;
    try {
      await client.httpTransport(
          'GET', probeUrl, {...jsonHeaders, 'x-project-key': rawKey}, null);
    } on VellumException catch (e) {
      if (e.statusCode != 401) rethrow;
      revokedKeyRejected = true;
      print('revokedKeyRejected: code=${e.code} status=${e.statusCode}');
    }
    if (!revokedKeyRejected) throw StateError('Revoked key was accepted');
  } catch (e) {
    print('auth example error: $e');
    rethrow;
  } finally {
    client.dispose();
  }
}

Map<String, dynamic> objectMap(Object? value, String context) {
  if (value is VellumHttpResponse) value = value.body;
  if (value is! Map<String, dynamic>)
    throw FormatException('Expected object: $context');
  return value;
}

/**
 * Authentication & API keys — full key lifecycle with immediate revocation.
 *
 * What you'll build: a sign-in, a session check, a server key creation, and a
 * verification that revoking that key takes effect on the very next request.
 * Real-world scenario: an operator rotates a leaked key. A freshly verified
 * key must stop authenticating immediately after it is revoked — no cache
 * window. All tokens/keys are kept out of output; only stable codes are logged.
 */
import { harness, check } from "./_harness.js";

const h = await harness("auth");
const keysPath = `/projects/${h.projectId}/api-keys`;

try {
  h.log("=== Authentication & API Keys ===");

  const login = await h.api.signIn();
  check(login.success === true && typeof login.token === "string" && login.token.length > 0, "sign-in returned a token");
  h.log({ signIn: { success: login.success, tokenLength: login.token.length, userRole: login.user["platformRole"] } });

  const session = await h.api.raw<{ session?: unknown }>(
    "GET",
    "/v1/auth/session",
    undefined,
    { authorization: `Bearer ${login.token}` }
  );
  check(session.status === 200, "session endpoint accepted the bearer token");
  h.log({ session: { status: session.status, hasSession: Boolean(session.body.session) } });

  const listed = await h.api.console<Array<{ id: string; keyType: string }>>("GET", keysPath);
  check(listed.ok && Array.isArray(listed.body) && listed.body.length > 0, "API keys are listable");
  h.log({ listKeys: { status: listed.status, nonEmpty: listed.body.length > 0, includesServerKey: listed.body.some((k) => k.keyType === "server") } });

  const created = await h.api.console<{
    apiKey?: { id: string; keyType: string };
    rawKey?: string;
  }>("POST", keysPath, { name: "examples:auth-ephemeral", keyType: "server" });
  check(created.status === 201 && Boolean(created.body.apiKey?.id) && Boolean(created.body.rawKey), "ephemeral key created");
  const ephemeralId = created.body.apiKey?.id;
  h.log({ createKey: { status: created.status, keyType: created.body.apiKey?.keyType, idReturned: Boolean(ephemeralId) } });

  const unauth = await h.api.raw(
    "GET",
    `/v1/projects/${h.projectId}/documents/${h.prefix}/probe`
  );
  const unauthCode = (unauth.body as { error?: { code?: string } })?.error?.code;
  check(unauth.status === 401 && unauthCode === "UNAUTHENTICATED", "missing key returns 401 UNAUTHENTICATED");
  h.log({ missingKey: { status: unauth.status, code: unauthCode } });

  const docProbe = `/v1/projects/${h.projectId}/documents/${h.prefix}/probe`;
  const whileActive = await h.api.raw("GET", docProbe, undefined, { "x-project-key": created.body.rawKey as string });
  check(whileActive.status !== 401, "freshly created key authenticates");

  const revoked = await h.api.console("DELETE", `${keysPath}/${ephemeralId}`);
  check(revoked.ok, "ephemeral key was revoked");
  h.log({ revokeKey: { status: revoked.status, success: (revoked.body as { success?: boolean } | null)?.success } });

  // Revocation must take effect immediately, even for a recently verified key.
  const afterRevoke = await h.api.raw("GET", docProbe, undefined, { "x-project-key": created.body.rawKey as string });
  const revokeCode = (afterRevoke.body as { error?: { code?: string } })?.error?.code;
  check(afterRevoke.status === 401 && revokeCode === "UNAUTHENTICATED", "revoked key is rejected immediately");
  h.log({ revokedKeyRejected: { status: afterRevoke.status, code: revokeCode } });
} finally {
  await h.cleanup();
}

  h.log("=== Authentication & API Keys ===");

  const login = await h.api.signIn();
  check(login.success === true && typeof login.token === "string" && login.token.length > 0, "sign-in returned a token");
  h.log({ signIn: { success: login.success, tokenLength: login.token.length, userRole: login.user.platformRole } });

  const session = await h.api.raw("GET", "/v1/auth/session", undefined, {
    authorization: `Bearer ${login.token}`,
  });
  check(session.status === 200, "session endpoint accepted the bearer token");
  h.log({ session: { status: session.status, hasSession: Boolean(session.body.session) } });

  const listed = await h.api.console("GET", keysPath);
  check(listed.ok && Array.isArray(listed.body) && listed.body.length > 0, "API keys are listable");
  h.log({
    listKeys: {
      status: listed.status,
      nonEmpty: listed.body.length > 0,
      includesServerKey: listed.body.some((k) => k.keyType === "server"),
    },
  });

  const created = await h.api.console("POST", keysPath, { name: "examples:rest-auth:ephemeral", keyType: "server" });
  check(created.status === 201 && Boolean(created.body.apiKey?.id) && Boolean(created.body.rawKey), "ephemeral key created");
  const ephemeralId = created.body.apiKey.id;
  h.log({
    createKey: { status: created.status, keyType: created.body.apiKey.keyType, idReturned: Boolean(ephemeralId) },
  });

  const unauth = await h.api.raw("GET", docProbe);
  check(unauth.status === 401 && unauth.body?.error?.code === "UNAUTHENTICATED", "missing key returns 401");
  h.log({ missingKey: { status: unauth.status, code: unauth.body.error.code } });

  const whileActive = await h.api.raw("GET", docProbe, undefined, { "x-project-key": created.body.rawKey });
  check(whileActive.status !== 401, "freshly created key authenticates");

  const revoked = await h.api.console("DELETE", `${keysPath}/${ephemeralId}`);
  check(revoked.ok, "ephemeral key revoked");
  h.log({ revokeKey: { status: revoked.status, success: revoked.body?.success } });

  const afterRevoke = await h.api.raw("GET", docProbe, undefined, { "x-project-key": created.body.rawKey });
  check(afterRevoke.status === 401 && afterRevoke.body?.error?.code === "UNAUTHENTICATED", "revoked key rejected immediately");
  h.log({ revokedKeyRejected: { status: afterRevoke.status, code: afterRevoke.body.error.code } });
} finally {
  await h.cleanup();
}
