المرجع / الأخطاء والحدود

الأخطاء والحدود#

تستخدم كل الإخفاقات غلافًا قياسيًا واحدًا ومجموعة واحدة من رموز الأخطاء، المعرّفة في @vellum/core (SYSTEM_LIMITS وVellumError). كل استجابة HTTP تحمل أيضًا الترويسة X-Request-Id.

السيناريو#

أنت تربط معالجة الأخطاء في العميل وتريد إعادة المحاولة فقط للإخفاقات العابرة: تعارض الإصدار يجب أن يُعيد القراءة والمحاولة، وحد المعدل يجب أن يتراجع باستخدام Retry-After، أما خطأ الصلاحيات فيجب أن يظهر للمستخدم. تُفرّع على الحقل code القياسي من غلاف الخطأ.

TS
import { VellumError } from "@vellum/core";

try {
  await db.collection("users").doc("u_123").update({ name: "Ada" }, { ifVersion: 4 });
} catch (err) {
  const e = err as { code?: string; status?: number; details?: Record<string, unknown> };
  switch (e.code) {
    case "FAILED_PRECONDITION": // أعد القراءة وأعد المحاولة بالإصدار الحالي
      break;
    case "ABORTED_CONFLICT": // أعد تشغيل المعاملة كاملة
      break;
    case "PERMISSION_DENIED": // تحقق من النطاقات أو القواعد
      break;
    default:
      throw err;
  }
}

غلاف الخطأ#

JSON
{
  "error": {
    "code": "FAILED_PRECONDITION",
    "message": "Version precondition failed on 'users/u_123' (expected 4, actual 5)",
    "requestId": "req_01JCK9X2P8...",
    "details": {
      "path": "users/u_123",
      "expectedVersion": 4,
      "actualVersion": 5
    }
  }
}

في SDK، تُثير الطلبات الفاشلة خطأ Error يحمل code وstatus وdetails وrequestId المنسوخة من الغلاف. الكائن details مكتوب كسجل (record) وتختلف مفاتيحه حسب الرمز؛ لا تعتمد على شكل ثابت.

  await new Promise((r) => setTimeout(r, retryAfterMs));

معالجة الأخطاء#

يُظهر المثال في بداية هذا الدليل التفريع على e.code. أما في عملاء HTTP الخام، فاقرأ حقل error.code من غلاف JSON والترويسة X-Request-Id.

TIP

أدرج X-Request-Id (أو error.requestId) دائمًا في تقارير الأخطاء. السجلات مُخفاة، لذا معرّف الطلب هو الطريقة الآمنة لربط حادثة ما دون مشاركة أسرار.

الأخطاء الشائعة#

  • التفريع على حالة HTTP وحدها. رموز متعددة تشترك في الحالة نفسها (مثل 400)؛ فَرِّع على code القياسي.
  • إعادة محاولة كل خطأ. فقط الرموز العابرة مثل ABORTED_CONFLICT وRATE_LIMITED آمنة لإعادة المحاولة؛ خطأ الصلاحيات لن يُصلح نفسه.
  • افتراض شكل ثابت لـ details. مفاتيحه تختلف حسب الرمز — عاملها كسجل.
  • تجاهل معرّف الطلب. بدونه لا يمكن للدعم ربط حادثة مُخفاة بطلبك.

مثال موثّق#

المجموعة / البروتوكول
DART
import 'dart:io';
// Executed against the live PostgreSQL API by scripts/run-dart-examples.mjs.
//
// Errors & limits — probe the canonical VELLUM error codes.
// What you'll build: a tour of quota/complexity/size/conflict errors, printing
// only stable code + status pairs.
// Real-world scenario: clients must handle these boundary failures gracefully.
// `VellumException` carries `code` and `statusCode` for every >=400 response.
import 'package:vellum_flutter/vellum_flutter.dart';

Future<void> main() async {
  final endpoint = Platform.environment['VELLUM_API_BASE'] ??
      const String.fromEnvironment('VELLUM_API_BASE',
          defaultValue: 'http://127.0.0.1:8080');
  final projectId = Platform.environment['VELLUM_PROJECT_ID'] ??
      const String.fromEnvironment('VELLUM_PROJECT_ID');
  final apiKey = Platform.environment['VELLUM_API_KEY'] ??
      const String.fromEnvironment('VELLUM_API_KEY');

  final client =
      VellumClient(endpoint: endpoint, projectId: projectId, apiKey: apiKey);
  final prefix = 'ex_flutter_errors_${DateTime.now().microsecondsSinceEpoch}';

  Future<void> probe(String label, Future<void> Function() action) async {
    try {
      await action();
      print('$label: unexpected success');
    } on VellumException catch (e) {
      print('$label: code=${e.code} status=${e.statusCode}');
    }
  }

  try {
    print('=== Errors & Limits (Flutter) ===');

    await probe('limit', () => client.collection(prefix).limit(501).get());

    Query<Map<String, dynamic>> complex = client.collection(prefix);
    for (var i = 0; i < 11; i++) {
      complex = complex.where('f$i', '==', i);
    }
    await probe('whereClauses=11', () => complex.get());

    final writes = List.generate(
        101,
        (i) => {
              'type': 'create',
              'path': '$prefix/bulk-$i',
              'data': {'i': i},
            });
    await probe('batchOperations=101',
        () => client.request('POST', '/batchWrite', body: {'writes': writes}));

    await client.request('POST', '/indexes', body: {
      'collectionPath': prefix,
      'fields': [
        {'path': 'onlyField', 'dir': 'asc'},
      ]
    });
    await probe(
        'duplicateIndex',
        () => client.request('POST', '/indexes', body: {
              'collectionPath': prefix,
              'fields': [
                {'path': 'onlyField', 'dir': 'asc'},
              ],
            }));

    await client.doc('$prefix/stale').set({'n': 0});
    await probe('staleVersion',
        () => client.doc('$prefix/stale').update({'n': 2}, ifVersion: 0));

    await probe(
        'documentTooLarge',
        () => client.request('POST', '/documents/$prefix/too-large', body: {
              'data': {'blob': 'x' * 1048577},
            }));

    await probe('notFound', () => client.doc('$prefix/does-not-exist').get());
  } finally {
    try {
      await client.doc('$prefix/stale').delete();
    } catch (_) {}
    client.dispose();
  }
}

/**
 * Errors & limits — probe the API's guardrails.
 *
 * What you'll build: a tour of the canonical Vellum error codes by deliberately
 * violating each documented quota.
 * Real-world scenario: when a client sends an over-limit query or an oversized
 * document, the app should show a useful message instead of crashing. This
 * example proves each boundary returns a stable `code` + HTTP `status` pair.
 * Only codes and statuses are printed — no secrets or request IDs.
 */
import { harness, check } from "./_harness.js";

const h = await harness("errors");
const col = h.prefix;

function codeOf(error: unknown): string {
  return (error as { code?: string }).code ?? "UNKNOWN";
}

try {
  h.log("=== Errors & Limits ===");

  try {
    await h.sdk.collection(col).limit(501).get();
    throw new Error("limit above 500 should fail");
  } catch (error: unknown) {
    check(codeOf(error) === "LIMIT_EXCEEDED", "MAX_QUERY_LIMIT returns LIMIT_EXCEEDED");
    h.log({ limit: codeOf(error), status: (error as { status?: number }).status });
  }

  let complex = h.sdk.collection(col);
  for (let i = 0; i < 11; i += 1) complex = complex.where(`f${i}`, "==", i);
  try {
    await complex.get();
    throw new Error("more than 10 where clauses should fail");
  } catch (error: unknown) {
    check(codeOf(error) === "QUERY_TOO_COMPLEX", "MAX_WHERE_CLAUSES returns QUERY_TOO_COMPLEX");
    h.log({ whereClauses: 11, code: codeOf(error) });
  }

  const oversizedBatch = Array.from({ length: 101 }, (_, i) => ({
    type: "create",
    path: `${col}/bulk-${i}`,
    data: { i },
  }));
  try {
    await h.sdk.batchWrite(oversizedBatch);
    throw new Error("more than 100 batch operations should fail");
  } catch (error: unknown) {
    check(codeOf(error) === "BATCH_SIZE_EXCEEDED", "MAX_BATCH_OPERATIONS returns BATCH_SIZE_EXCEEDED");
    h.log({ batchOperations: 101, code: codeOf(error) });
  }

  const firstIndex = await h.api.data("POST", "/indexes", {
    collectionPath: col,
    fields: [{ path: "onlyField", dir: "asc" }],
  });
  const duplicateIndex = await h.api.data("POST", "/indexes", {
    collectionPath: col,
    fields: [{ path: "onlyField", dir: "asc" }],
  });
  const duplicateCode = (duplicateIndex.body as { error?: { code?: string } })?.error?.code;
  check(firstIndex.status === 201 && duplicateIndex.status === 409, "duplicate index returns 409");
  check(duplicateCode === "ALREADY_EXISTS", "duplicate index code is ALREADY_EXISTS");
  h.log({ duplicateIndex: { status: duplicateIndex.status, code: duplicateCode } });

  await h.sdk.collection(col).doc("stale").set({ n: 0 });
  try {
    await h.sdk.collection(col).doc("stale").update({ n: 2 }, { ifVersion: 0 });
    throw new Error("stale version should fail");
  } catch (error: unknown) {
    check(codeOf(error) === "FAILED_PRECONDITION", "stale version returns FAILED_PRECONDITION");
    h.log({ staleVersion: { code: codeOf(error), status: (error as { status?: number }).status } });
  }

  const tooLarge = await h.api.data("POST", `/documents/${col}/too-large`, {
    data: { blob: "x".repeat(1_048_577) },
  });
  const tooLargeCode = (tooLarge.body as { error?: { code?: string } })?.error?.code;
  check(tooLarge.status === 413 && tooLargeCode === "DOCUMENT_TOO_LARGE", "1 MiB+ doc returns DOCUMENT_TOO_LARGE");
  h.log({ documentTooLarge: { status: tooLarge.status, code: tooLargeCode } });

  const missing = await h.api.data("GET", `/documents/${col}/does-not-exist`);
  check(missing.status === 404, "missing document returns 404");
  h.log({ notFound: { status: missing.status, code: (missing.body as { error?: { code?: string } })?.error?.code } });
} finally {
  await h.cleanup();
}

/**
 * Errors & limits (REST) — probe the API's guardrails with fetch.
 *
 * What you'll build: a tour of canonical error codes by deliberately breaking
 * documented quotas, printing only stable code + status pairs.
 * Real-world scenario: clients must handle limit, complexity, size and conflict
 * errors gracefully; this example proves each boundary's exact canonical code.
 */
import { harness, check } from "./_harness.mjs";

const h = await harness("errors");
const col = h.prefix;
const codeOf = (reply) => reply.body?.error?.code ?? "UNKNOWN";

try {
  h.log("=== Errors & Limits ===");

  const overLimit = await h.api.data("POST", "/query", {
    collectionPath: col,
    where: [],
    orderBy: [],
    limit: 501,
    select: null,
    startAt: null,
    startAfter: null,
  });
  check(overLimit.status === 400 && codeOf(overLimit) === "LIMIT_EXCEEDED", "limit above 500 rejected");
  h.log({ limit: codeOf(overLimit), status: overLimit.status });

  const where = Array.from({ length: 11 }, (_, i) => ({ field: `f${i}`, op: "==", value: i }));
  const complex = await h.api.data("POST", "/query", {
    collectionPath: col,
    where,
    orderBy: [],
    limit: 50,
    select: null,
    startAt: null,
    startAfter: null,
  });
  check(complex.status === 400 && codeOf(complex) === "QUERY_TOO_COMPLEX", "11 where clauses rejected");
  h.log({ whereClauses: 11, code: codeOf(complex) });

  const writes = Array.from({ length: 101 }, (_, i) => ({
    type: "create",
    path: `${col}/bulk-${i}`,
    data: { i },
  }));
  const oversizedBatch = await h.api.data("POST", "/batchWrite", { writes });
  check(oversizedBatch.status === 400 && codeOf(oversizedBatch) === "BATCH_SIZE_EXCEEDED", "101 writes rejected");
  h.log({ batchOperations: 101, code: codeOf(oversizedBatch) });

  const firstIndex = await h.api.data("POST", "/indexes", {
    collectionPath: col,
    fields: [{ path: "onlyField", dir: "asc" }],
  });
  const duplicateIndex = await h.api.data("POST", "/indexes", {
    collectionPath: col,
    fields: [{ path: "onlyField", dir: "asc" }],
  });
  check(firstIndex.status === 201 && duplicateIndex.status === 409, "duplicate index returns 409");
  check(codeOf(duplicateIndex) === "ALREADY_EXISTS", "duplicate index code is ALREADY_EXISTS");
  h.log({ duplicateIndex: { status: duplicateIndex.status, code: codeOf(duplicateIndex) } });

  await h.api.data("POST", `/documents/${col}/stale`, { data: { n: 0 } });
  const stale = await h.api.data("PATCH", `/documents/${col}/stale`, { data: { n: 2 }, ifVersion: 0 });
  check(stale.status === 412 && codeOf(stale) === "FAILED_PRECONDITION", "stale version rejected");
  h.log({ staleVersion: { code: codeOf(stale), status: stale.status } });

  const tooLarge = await h.api.data("POST", `/documents/${col}/too-large`, {
    data: { blob: "x".repeat(1_048_577) },
  });
  check(tooLarge.status === 413 && codeOf(tooLarge) === "DOCUMENT_TOO_LARGE", "oversized doc rejected");
  h.log({ documentTooLarge: { status: tooLarge.status, code: codeOf(tooLarge) } });

  const missing = await h.api.data("GET", `/documents/${col}/does-not-exist`);

} finally {
  await h.cleanup();
}
